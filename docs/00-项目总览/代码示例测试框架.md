## 4. 测试示例

### 4.1 Python测试示例

**examples/python/test_lru_cache.py**：

```python
import pytest
from lru_cache import LRUCache

class TestLRUCache:
    """LRU缓存测试类"""

    def test_basic_operations(self):
        """测试基本操作"""
        cache = LRUCache(2)

        # 测试插入
        cache.put(1, 1)
        cache.put(2, 2)
        assert cache.get(1) == 1

        # 测试淘汰
        cache.put(3, 3)
        assert cache.get(2) == -1  # 2被淘汰

    def test_capacity(self):
        """测试容量限制"""
        cache = LRUCache(2)
        cache.put(1, 1)
        cache.put(2, 2)
        cache.put(3, 3)

        assert cache.get(1) == -1  # 1被淘汰
        assert cache.get(2) == 2
        assert cache.get(3) == 3

    def test_update_existing_key(self):
        """测试更新已存在的键"""
        cache = LRUCache(2)
        cache.put(1, 1)
        cache.put(1, 2)

        assert cache.get(1) == 2
```

### 4.2 Java测试示例

**examples/java/LRUCacheTest.java**：

```java
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.BeforeEach;
import static org.junit.jupiter.api.Assertions.*;

public class LRUCacheTest {
    private LRUCache cache;

    @BeforeEach
    void setUp() {
        cache = new LRUCache(2);
    }

    @Test
    void testBasicOperations() {
        cache.put(1, 1);
        cache.put(2, 2);
        assertEquals(1, cache.get(1));
    }

    @Test
    void testEviction() {
        cache.put(1, 1);
        cache.put(2, 2);
        cache.put(3, 3);
        assertEquals(-1, cache.get(1)); // 1被淘汰
    }
}
```

### 4.3 Go测试示例

**examples/go/lru_cache_test.go**：

```go
package main

import (
    "testing"
)

func TestLRUCache(t *testing.T) {
    cache := NewLRUCache(2)

    cache.Put(1, 1)
    cache.Put(2, 2)

    if val := cache.Get(1); val != 1 {
        t.Errorf("Expected 1, got %d", val)
    }

    cache.Put(3, 3)
    if val := cache.Get(2); val != -1 {
        t.Errorf("Expected -1 (evicted), got %d", val)
    }
}
```

## 5. CI/CD集成

### 5.1 GitHub Actions工作流

**.github/workflows/test-examples.yml**：

```yaml
name: Test Examples

on:
  push:
    branches:
      - main
    paths:
      - 'examples/**'
  pull_request:
    branches:
      - main
    paths:
      - 'examples/**'

jobs:
  test-python:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-python@v4
        with:
          python-version: '3.x'
      - name: Install dependencies
        run: pip install pytest
      - name: Run tests
        run: pytest examples/python/

  test-java:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-java@v3
        with:
          java-version: '17'
      - name: Run tests
        run: |
          cd examples/java
          ./gradlew test

  test-go:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-go@v4
        with:
          go-version: '1.21'
      - name: Run tests
        run: go test ./examples/go/...
```
